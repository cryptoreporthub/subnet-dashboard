```yaml
name: ci-smoke
on: [push]
jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        shard-index: [0, 1, 2]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - run: pip install pytest pytest-shard
      - run: python -m pytest -q tests/test_r27_shard_smoke.py --shard-id=${{ matrix.shard-index }} --num-shards=3
```
